cmake_minimum_required(VERSION 3.20)
project(ari_lyrics VERSION 1.7.9 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Core DBus Network)
add_library(lyrics_core STATIC lyrics.cpp catalog.cpp playback.cpp fullscreen.cpp artwork.cpp)
target_link_libraries(lyrics_core PUBLIC Qt6::Core Qt6::Network)
target_compile_options(lyrics_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(ari-lyrics main.cpp)
set_target_properties(ari-lyrics PROPERTIES AUTOMOC ON)
target_link_libraries(ari-lyrics PRIVATE lyrics_core Qt6::DBus Qt6::Network)
target_compile_options(ari-lyrics PRIVATE -Wall -Wextra -Wpedantic)
add_executable(lyrics-tests tests.cpp)
target_link_libraries(lyrics-tests PRIVATE lyrics_core)
enable_testing()
add_test(NAME lyrics-core COMMAND lyrics-tests)
add_executable(mpris-tests integration.cpp)
set_target_properties(mpris-tests PROPERTIES AUTOMOC ON)
target_link_libraries(mpris-tests PRIVATE Qt6::Core Qt6::DBus Qt6::Network)
add_test(NAME mpris-integration COMMAND dbus-run-session -- $<TARGET_FILE:mpris-tests> $<TARGET_FILE:ari-lyrics>)
install(TARGETS ari-lyrics RUNTIME DESTINATION bin)
install(DIRECTORY runtime/ DESTINATION . FILES_MATCHING PATTERN "*.luau" PATTERN "plugin.toml")

add_library(spectrum_core STATIC spectrum.cpp)
target_include_directories(spectrum_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(spectrum_core PRIVATE -Wall -Wextra -Wpedantic)
add_executable(spectrum-tests tests/spectrum-tests.cpp)
target_link_libraries(spectrum-tests PRIVATE spectrum_core)
add_test(NAME spectrum-core COMMAND spectrum-tests)
add_executable(artwork-tests tests/artwork-tests.cpp)
target_include_directories(artwork-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
target_link_libraries(artwork-tests PRIVATE lyrics_core)
add_test(NAME artwork-core COMMAND artwork-tests)

find_package(PkgConfig QUIET)
if(PkgConfig_FOUND)
    pkg_check_modules(PIPEWIRE QUIET IMPORTED_TARGET libpipewire-0.3)
endif()
if(PIPEWIRE_FOUND)
    add_executable(ari-lyrics-spectrum spectrum-main.cpp)
    target_link_libraries(ari-lyrics-spectrum PRIVATE spectrum_core Qt6::Core PkgConfig::PIPEWIRE)
    target_compile_options(ari-lyrics-spectrum PRIVATE -Wall -Wextra)
    install(TARGETS ari-lyrics-spectrum RUNTIME DESTINATION bin)
else()
    message(STATUS "libpipewire-0.3 not found: native spectrum disabled")
endif()

find_program(LUA_EXECUTABLE lua)
if(LUA_EXECUTABLE)
    foreach(check island widget player panel motion lyric-scroll search)
        add_test(NAME ui-${check} COMMAND ${LUA_EXECUTABLE} verify-${check}.lua)
        set_tests_properties(ui-${check} PROPERTIES WORKING_DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/runtime)
    endforeach()
endif()
find_package(Python3 3.11 QUIET COMPONENTS Interpreter)
if(Python3_Interpreter_FOUND)
    add_test(NAME keybindings-config COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/keybindings-tests.py)
    add_test(NAME installer-config COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/install-tests.py)
    set_tests_properties(installer-config PROPERTIES ENVIRONMENT "PYTHONDONTWRITEBYTECODE=1")
    if(TARGET ari-lyrics-spectrum)
        add_test(NAME spectrum-stream COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/spectrum-stream-tests.py $<TARGET_FILE:ari-lyrics-spectrum>)
        add_test(NAME spectrum-restart COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/spectrum-restart-tests.py $<TARGET_FILE:ari-lyrics-spectrum>)
        set_tests_properties(spectrum-restart PROPERTIES SKIP_RETURN_CODE 77)
    endif()
endif()
